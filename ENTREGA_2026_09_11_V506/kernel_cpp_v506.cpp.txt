/// Verifica que el sistema operativo ha habilitado los registros ZMM de 512
/// bits en el Extended Control Register (XCR0). Sin esta verificaciÃ³n, usar
/// instrucciones AVX-512 en un OS que no las soporta genera SIGILL.
///
/// Bits requeridos en XCR0:
///   bit 1 = SSE state (XMM)
///   bit 2 = AVX state (YMM)
///   bit 5 = opmask (k0-k7)
///   bit 6 = ZMM_Hi256 (parte alta de ZMM0-15)
///   bit 7 = Hi16_ZMM (ZMM16-31)
///   MÃ¡scara combinada: 0xE6
static bool os_supports_avx512() {
#if defined(__x86_64__) || defined(_M_X64) || defined(__i386__) || defined(_M_IX86)
  #if defined(_MSC_VER)
    int info[4] = {0};
    __cpuid(info, 1);
    if (!(info[2] & (1 << 27))) return false; // OSXSAVE no disponible
    unsigned long long xcr0 = _xgetbv(0);
  #else
    unsigned int a, b, c, d;
    if (!__get_cpuid(1, &a, &b, &c, &d)) return false;
    if (!(c & (1 << 27))) return false; // OSXSAVE no disponible
    unsigned int lo, hi;
    __asm__ __volatile__("xgetbv" : "=a"(lo), "=d"(hi) : "c"(0));
    unsigned long long xcr0 = ((unsigned long long)hi << 32) | lo;
  #endif
  return (xcr0 & 0xE6) == 0xE6;
#else
  return false;
#endif
}

/// Interroga CPUID para determinar el ancho SIMD mÃ¡ximo soportado por el CPU
/// Y por el OS, y retorna el alineamiento Ã³ptimo en bytes:
///   64 = AVX-512 (ZMM, 512 bits)
///   32 = AVX2 o AVX (YMM, 256 bits)
///   16 = SSE (XMM, 128 bits) o ARM NEON
///
/// FIX #33: No basta con verificar CPUID leaf 7. El OS debe haber habilitado
/// los registros extendidos vÃ­a XGETBV, o las instrucciones generan SIGILL.
static int detect_alignment() {
#if defined(__x86_64__) || defined(_M_X64) || defined(__i386__) || defined(_M_IX86)
  #if defined(_MSC_VER)
    int info[4] = {0};
    __cpuidex(info, 7, 0);
    // AVX-512F = leaf 7, EBX bit 16. Requiere XGETBV verificado.
    if ((info[1] & (1 << 16)) && os_supports_avx512()) return 64;
    // AVX2 = leaf 7, EBX bit 5
    if (info[1] & (1 << 5)) return 32;
    // AVX = leaf 1, ECX bit 28
    __cpuid(info, 1);
    if (info[2] & (1 << 28)) return 32;
  #else
    unsigned int a, b, c, d;
    if (__get_cpuid_count(7, 0, &a, &b, &c, &d)) {
      if ((b & (1 << 16)) && os_supports_avx512()) return 64;
      if (b & (1 << 5)) return 32;
    }
    if (__get_cpuid(1, &a, &b, &c, &d)) {
      if (c & (1 << 28)) return 32;
    }
  #endif
  return 16; // SSE2 mÃ­nimo garantizado en x86_64
#elif defined(__aarch64__) || defined(_M_ARM64)
  return 16; // NEON es 128 bits en AArch64
#else
  return 16;
#endif
}

/// Cache del resultado de detect_alignment() (se ejecuta una sola vez).
static int cached_alignment() {
  static int a = detect_alignment();
  return a;
}

/// Consulta el tamaÃ±o de la lÃ­nea de cachÃ© L1 de DATOS del CPU actual.
///
/// FIX #34: En Linux, /sys/devices/system/cpu/cpu0/cache/index0 puede ser
/// la cachÃ© de INSTRUCCIONES (type=Instruction), que tiene un tamaÃ±o de
/// lÃ­nea diferente en algunos ARM y x86. Solo aceptamos type=Data o Unified.
///
/// En Windows, usamos GetLogicalProcessorInformation con CacheData filter.
///
/// Retorna 64 como fallback conservador si no se puede determinar.
PMTP_EXPORT int pmtp_cpp_query_cache_line_size() {
#if defined(_WIN32)
  SYSTEM_LOGICAL_PROCESSOR_INFORMATION buf[256];
  DWORD len = sizeof(buf);
  if (GetLogicalProcessorInformation(buf, &len)) {
    DWORD n = len / sizeof(SYSTEM_LOGICAL_PROCESSOR_INFORMATION);
    for (DWORD i = 0; i < n; i++) {
      if (buf[i].Relationship == RelationCache
          && buf[i].Cache.Level == 1
          && buf[i].Cache.Type == CacheData) {
        return static_cast<int>(buf[i].Cache.LineSize);
      }
    }
  }
  return 64;
#elif defined(__linux__)
  // Iterar sobre los indices de cachÃ© buscando type=Data o Unified
  for (int idx = 0; idx < 8; idx++) {
    char path[256], type_str[32] = {0};
    snprintf(path, sizeof(path),
             "/sys/devices/system/cpu/cpu0/cache/index%d/type", idx);
    FILE* f = fopen(path, "r");
    if (!f) continue;
    if (fscanf(f, "%31s", type_str) != 1) { fclose(f); continue; }
    fclose(f);
    // Solo aceptar Data o Unified (no Instruction)
    if (strcmp(type_str, "Data") != 0 && strcmp(type_str, "Unified") != 0)
      continue;
    snprintf(path, sizeof(path),
             "/sys/devices/system/cpu/cpu0/cache/index%d/coherency_line_size",
             idx);
    f = fopen(path, "r");
    if (!f) continue;
    int line = 0;
    if (fscanf(f, "%d", &line) != 1) { fclose(f); continue; }
    fclose(f);
    if (line > 0) return line;
  }
  return 64;
#else
  return 64;
#endif
}

/// Valida que un tensor en memoria cumple los requisitos de alineamiento
/// SIMD detectados en tiempo de ejecuciÃ³n y que su tamaÃ±o es suficiente.
///
/// Retorna 1 si vÃ¡lido, cÃ³digo negativo si no:
///  -1 = puntero NULL, -2 = tamaÃ±o 0, -3 = min_dim â‰¤ 0,
///  -4 = desalineado, -5 = tamaÃ±o insuficiente.
PMTP_EXPORT int pmtp_cpp_validate_tensor_alignment(
    const void* tensor_ptr, uint64_t size_bytes, int min_dim) {
  if (!tensor_ptr) return -1;
  if (size_bytes == 0) return -2;
  if (min_dim <= 0) return -3;
  int req = cached_alignment();
  if ((reinterpret_cast<uintptr_t>(tensor_ptr) % req) != 0) return -4;
  if (size_bytes < static_cast<uint64_t>(min_dim) * sizeof(float)) return -5;
  return 1;
}

/// Retorna el alineamiento SIMD Ã³ptimo detectado (16, 32, o 64 bytes).
PMTP_EXPORT int pmtp_cpp_query_alignment_requirement() {
  return cached_alignment();
}
